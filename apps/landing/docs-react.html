React & Next.js SDK

Integrate feature flags seamlessly into your component tree. 

import { FlagProvider, useFlag } from 'flagcraft/react';

function App() {
  return (
    <FlagProvider apiKey="env_123" userId="user_abc">
      <Checkout />
    </FlagProvider>
  );
}

function Checkout() {
  const isNewFlowEnabled = useFlag('new-checkout-flow');
  return isNewFlowEnabled ? <NewCheckout /> : <OldCheckout />;
}
